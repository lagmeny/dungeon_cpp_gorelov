cmake_minimum_required(VERSION 3.10)
project(dungeon_practices_9_10 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(dungeon main.cpp Map.cpp Hero.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(dungeon PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME lifetime COMMAND dungeon --demo-lifetime)
